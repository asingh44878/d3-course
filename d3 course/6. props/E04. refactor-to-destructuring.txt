export default function Badge ({size, color, label}) {   --> props are passed within single culry brackets.

        height: size,
        borderRadius: "50%",
        backgroundColor: color,
        display: "flex",
        alignItems: "center",
        justifyContent: "center",
        color: "white",
        fontWeight: "bold",
          fontSize: size / 3,
      }}
    >
      {label}
    </div>
  );
};
