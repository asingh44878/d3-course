-- Scatterplot.jsx

export default function Scatterplot ({data}) {
    return(
        <div style={{position: "relative", width: 300, height: 120}}>
            {data.map((d, i) => (
                <div
                    key={i}
                    style={{
                        position: "absolute",
                        left: d.x,
                        top: d.y,
                        width: 16,
                        height: 16,
                        borderRadius: "50%",
                        backgroundColor: "dodgerblue"
                    }}
                </div>
            ))}
        </div>
    );
};

-- App.js

import Scatterplot from "./Scatterplot";

const data = [
    {x: 30, y:20},
    {x: 80, y:90},
    {x: 150, y:10},
    {x: 200, y:60},

export defualt App (data) {
